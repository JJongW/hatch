```ts
export type UsageFeature = 'doc' | 'questions' | 'tool';
/** 호출하는 쪽이 넘기는 꼬리표. key = DocKey · 'questions' · ToolKey */

export type UsageEntry = UsageTag & {
  v: 1;
  /** ISO 8601, 호출이 끝난 시각 */
  at: string;
  ok: boolean;
  ms: number;
  /** num_turns. 1보다 크면 모델이 여러 번 돎 */
  turns: number | null;
  input: number | null;
  cacheWrite: number | null;
  cacheRead: number | null;
  output: number | null;
  /** total_cost_usd — 정가 기준 추정, 구독은 실제 청구 아님 */
  costUSD: number | null;
  model: string | null;
  /** 프롬프트 길이만. 내용은 남기지 않는다 */
  promptChars: number;
};
```

```ts
import 'server-only';

import fs from 'node:fs/promises';
import path from 'node:path';

import { dataDir, isMissing } from './paths';
import { USAGE_FILE_VERSION, type UsageEntry } from './types';

const usageFile = () => path.join(dataDir(), 'usage.jsonl');

/** 호출 한 번을 한 줄로 덧붙인다. 실패해도 던지지 않는다 — 장부 때문에 생성이 막히면 안 된다 */
export async function appendUsage(entry: Omit<UsageEntry, 'v'>): Promise<void> {
  try {
    await fs.mkdir(dataDir(), { recursive: true });
    await fs.appendFile(usageFile(), `${JSON.stringify({ v: USAGE_FILE_VERSION, ...entry })}\n`);
  } catch (e) {
    console.error('[hatch] usage.jsonl', e);
  }
}

const isEntry = (x: unknown): x is UsageEntry => {
  const e = x as Partial<UsageEntry> | null;
  return (
    !!e && e.v === USAGE_FILE_VERSION && typeof e.feature === 'string' && typeof e.key === 'string' &&
    typeof e.at === 'string' && typeof e.ok === 'boolean' && typeof e.ms === 'number'
  );
};

/** 장부 전체. 깨진 줄은 건너뛰고 개수만 알린다 (1MB를 넘으면 읽기 속도를 보고 월별로 나눈다) */
export async function listUsage(): Promise<{ entries: UsageEntry[]; unreadable: number }> {
  let raw: string;
  try {
    raw = await fs.readFile(usageFile(), 'utf8');
  } catch (e) {
    if (isMissing(e)) return { entries: [], unreadable: 0 };
    throw e;
  }
  const entries: UsageEntry[] = [];
  let unreadable = 0;
  for (const line of raw.split('\n')) {
    if (!line.trim()) continue;
    try {
      const x: unknown = JSON.parse(line);
      if (isEntry(x)) entries.push(x);
      else unreadable += 1;
    } catch {
      unreadable += 1;
    }
  }
  return { entries, unreadable };
}
```

```ts
import type { UsageEntry } from '@/lib/storage';

export type RunStats = Pick<UsageEntry, 'turns' | 'input' | 'cacheWrite' | 'cacheRead' | 'output' | 'costUSD' | 'model'>;

const EMPTY: RunStats = { turns: null, input: null, cacheWrite: null, cacheRead: null, output: null, costUSD: null, model: null };
const num = (x: unknown) => (typeof x === 'number' && Number.isFinite(x) ? x : null);

/** claude -p --output-format json 결과에서 사용량 숫자만 꺼낸다. 결과가 없거나 깨졌으면 모두 null */
export function statsOf(stdout: string | null): RunStats {
  if (!stdout) return EMPTY;
  let r: { num_turns?: unknown; total_cost_usd?: unknown; usage?: Record<string, unknown>; modelUsage?: Record<string, unknown> };
  try {
    r = JSON.parse(stdout);
  } catch {
    return EMPTY;
  }
  const u = r.usage ?? {};
  return {
    turns: num(r.num_turns),
    input: num(u.input_tokens),
    cacheWrite: num(u.cache_creation_input_tokens),
    cacheRead: num(u.cache_read_input_tokens),
    output: num(u.output_tokens),
    costUSD: num(r.total_cost_usd),
    model: Object.keys(r.modelUsage ?? {})[0] ?? null,
  };
}
```

```ts
// 사용량 장부를 화면용 숫자로 바꾼다. AI는 판단, 코드는 계산 — 합계, 평균, 신호는 여기서만
import type { UsageEntry, UsageFeature } from '@/lib/storage';

/** 신호 기준 (spec §4). 숫자를 바꾸려면 여기만 */
export const SIGNAL = { heavyShare: 0.3, redoMin: 2, redoShare: 0.5 } as const;

export type Signal = 'heavy' | 'loops' | 'redo' | 'fails';
export type FeatureRow = {
  feature: UsageFeature;
  key: string;
  calls: number;
  avgMs: number;
  avgTokens: number | null;
  avgOutput: number | null;
  avgCost: number | null;
  cost: number;
  costShare: number;
  fails: number;
  again: number;
  loops: number;
  signals: Signal[];
};
export type IdeaRow = { ideaId: string; calls: number; tokens: number; cost: number; lastAt: string };
export type UsageSummary = {
  calls: number;
  tokens: number;
  cost: number;
  from: string | null;
  to: string | null;
  features: FeatureRow[];
  ideas: IdeaRow[];
};

const hasTokens = (e: UsageEntry) => e.input !== null || e.cacheWrite !== null || e.cacheRead !== null || e.output !== null;
export const totalTokens = (e: UsageEntry) => (e.input ?? 0) + (e.cacheWrite ?? 0) + (e.cacheRead ?? 0) + (e.output ?? 0);
const avg = (xs: number[]) => (xs.length ? xs.reduce((a, b) => a + b, 0) / xs.length : null);
const sum = (xs: number[]) => xs.reduce((a, b) => a + b, 0);

export function summarizeUsage(all: UsageEntry[], ideaId?: string): UsageSummary {
  const entries = ideaId === undefined ? all : all.filter((e) => e.ideaId === ideaId);
  const cost = sum(entries.map((e) => e.costUSD ?? 0));
  const times = entries.map((e) => e.at).sort();

  const groups = new Map<string, UsageEntry[]>();
  for (const e of entries) {
    const k = `${e.feature}:${e.key}`;
    groups.set(k, [...(groups.get(k) ?? []), e]);
  }
  const many = groups.size >= 2;
  const features: FeatureRow[] = [...groups.values()].map((g) => {
    const measured = g.filter(hasTokens);
    const priced = g.filter((e) => e.costUSD !== null);
    const fCost = sum(g.map((e) => e.costUSD ?? 0));
    const costShare = cost > 0 ? fCost / cost : 0;
    const fails = g.filter((e) => !e.ok).length;
    const again = g.filter((e) => e.again).length;
    const loops = g.filter((e) => (e.turns ?? 0) > 1).length;
    const signals: Signal[] = [];
    if (many && costShare >= SIGNAL.heavyShare) signals.push('heavy');
    if (loops > 0) signals.push('loops');
    if (g.length >= SIGNAL.redoMin && again / g.length >= SIGNAL.redoShare) signals.push('redo');
    if (fails > 0) signals.push('fails');
    return {
      feature: g[0].feature,
      key: g[0].key,
      calls: g.length,
      avgMs: avg(g.map((e) => e.ms)) ?? 0,
      avgTokens: avg(measured.map(totalTokens)),
      avgOutput: avg(measured.filter((e) => e.output !== null).map((e) => e.output!)),
      avgCost: avg(priced.map((e) => e.costUSD!)),
      cost: fCost,
      costShare,
      fails,
      again,
      loops,
      signals,
    };
  });
  features.sort((a, b) => b.cost - a.cost || b.calls - a.calls);

  const byIdea = new Map<string, UsageEntry[]>();
  for (const e of entries) if (e.ideaId) byIdea.set(e.ideaId, [...(byIdea.get(e.ideaId) ?? []), e]);
  const ideas: IdeaRow[] = [...byIdea.entries()].map(([id, g]) => ({
    ideaId: id,
    calls: g.length,
    tokens: sum(g.map(totalTokens)),
    cost: sum(g.map((e) => e.costUSD ?? 0)),
    lastAt: g.map((e) => e.at).sort().at(-1)!,
  }));
  ideas.sort((a, b) => b.cost - a.cost || b.calls - a.calls);

  return {
    calls: entries.length,
    tokens: sum(entries.map(totalTokens)),
    cost,
    from: times[0] ?? null,
    to: times.at(-1) ?? null,
    features,
    ideas,
  };
}

const one = (n: number) => (Math.round(n * 10) / 10).toString();
export function formatTokens(n: number): string {
  if (n >= 10_000) return `${one(n / 10_000)}만`;
  if (n >= 1_000) return `${one(n / 1_000)}천`;
  return String(Math.round(n));
}
export function formatCost(usd: number): string {
  if (usd === 0) return '0달러';
  if (usd < 0.005) return '0.01달러 미만';
  return `${usd.toFixed(2)}달러`;
}
```

```tsx
import { connection } from 'next/server';

import { DOC_STEPS, isValidId, listIdeas, listUsage } from '@/lib/storage';
import { TOOLS } from '@/lib/tools';
import { summarizeUsage } from '@/lib/usage/summary';

import UsageView from './usage-view';

// 로컬 전용 · 항상 요청 시 렌더라 정적 셸 검증을 끈다 (docs: route-segment-config/instant, 'Disabling instant')
export const instant = false;

export default async function UsagePage({ searchParams }: { searchParams: Promise<{ idea?: string }> }) {
  await connection();
  const [{ entries, unreadable }, { ideas }, { idea }] = await Promise.all([
    listUsage(),
    listIdeas().catch(() => ({ ideas: [], unreadable: 0 })),
    searchParams,
  ]);
  const titles = Object.fromEntries(ideas.map((i) => [i.id, i.title]));
  const ideaId = idea && isValidId(idea) ? idea : undefined;
  // 기능 이름: 문서는 단계 이름, 도구는 도구 이름 (key가 겹칠 일은 없지만 feature로 구분해 둔다)
  const labels: Record<string, string> = {
    'questions:questions': '확인 질문',
    ...Object.fromEntries(DOC_STEPS.map((s) => [`doc:${s.key}`, s.label])),
    ...Object.fromEntries(Object.values(TOOLS).map((t) => [`tool:${t!.key}`, t!.label])),
  };
  return (
    <UsageView
      summary={summarizeUsage(entries, ideaId)}
      labels={labels}
      titles={titles}
      ideaFilter={ideaId ? { id: ideaId, title: titles[ideaId] ?? ideaId } : null}
      unreadable={unreadable}
    />
  );
}
```

```md
# 기능마다 토큰과 비용을 기록하고 비교한다

**날짜**: 2026-10-30
**관련 파일**: usage.ts, summary.ts, usage-view.tsx, run.ts

## What — 무엇이 달라졌나
문서를 만들거나 도구를 돌릴 때마다 걸린 시간, 토큰, 정가 기준 비용이 남는다. 사용량 화면에서 기능별 평균과 비중, 아이디어별 합계를 보고, 비용이 큰 기능이나 여러 번 도는 호출, 다시 만들기가 잦은 기능에 표시가 붙는다. 아이디어 상세에는 그 아이디어에 쓴 합계가 한 줄 보인다.

## Why — 왜 필요했나
어느 기능이 얼마나 드는지, 효율적으로 쓰는지 알 수 없었다. 전체 실제 테스트에서 PRD 리뷰 위원회 한 번이 전체 비용의 36%를 쓴다는 걸 손으로 재서야 알았다. Vercel로 옮길 때 API 요금을 낼지 정하려면 아이디어 하나에 드는 총비용도 알아야 했다.

## 판단 근거
- 실패와 다시 만들기까지 남겨야 낭비가 보인다. 그래서 문서 머리말이 아니라 호출마다 한 줄씩 쌓는 장부로 했다.
- 비용은 claude가 알려 주는 정가 기준 추정이다. 구독은 실제로 청구되지 않는다는 말을 늘 같이 보여 준다.
- 신호는 코드가 정한 기준으로만 붙인다. 입력 대비 출력이 적다는 신호는 기준을 정할 근거가 없어 넣지 않았다.
- 데모의 숫자는 2026-10-30 지어낸 아이디어로 실제 실행한 결과를 데모 아이디어에 붙인 것이다.
```
